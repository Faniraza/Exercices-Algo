*Algorithme Verification Palindrome*

Variables :
	mot : chaine
	inverse : chaine
	caractere : caractere
Début:
	Afficher("Entrez un mot :")
	Lire(mot)
	inverse <- ""
	Pour chaque caractere dans mot
		inverse <- caractere + inverse
	FinPour
	Si mot == inverse Alors
		Afficher("C'est un palindrome")
	Sinon
		Afficher("Ce n'est pas un palindrome")
	FinSi
Fin
