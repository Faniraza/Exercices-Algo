VARIABLES
    n REEL
    i REEL
    a REEL
    b REEL
    suivant REEL

DEBUT
    a <- 0
    b <- 1
    suivant <- 0
    i <- 0

    AFFICHER "Combien de termes de Fibonacci afficher ?  "
    LIRE n

    SI (n >= 1) ALORS
        AFFICHER a
        AFFICHER " "
    FIN_SI

    SI (n >= 2) ALORS
        AFFICHER b
        AFFICHER " "
    FIN_SI

    POUR (i <- 3, i <= n, i<-i+1) ALORS
        suivant <- a + b
        AFFICHER suivant
        AFFICHER " "
        a <- b
        b <- suivant
    FINPOUR
FIN
