*Algorithme  Chercher une Lettre*

Variables :
	texte : chaine
	cible : caracteres
	trouvé : booleen
	caractere : caractere

Début
    Lire(texte)
    Lire(cible)
    trouve <- Faux
    Pour chaque caractere dans texte :
        Si caractere == cible Alors
            trouvé <- Vrai
        FinSi
   FinPour
    	Si trouvé == Vrai Alors 
    		Afficher("La cible est dans la chaine")
	Sinon
		Afficher("Absent")
Fin
